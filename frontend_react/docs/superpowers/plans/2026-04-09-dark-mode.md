# Dark Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add dark mode support with system preference detection and a manual Light/Dark/System toggle in the user dropdown menu.

**Architecture:** CSS variable overrides via a `.dark` class on `<html>`, managed by a Zustand store with localStorage persistence. A flash-prevention script in `index.html` applies the class before React renders. All existing semantic CSS variables get dark counterparts; hardcoded Tailwind color classes are migrated to semantic equivalents where needed.

**Tech Stack:** Tailwind CSS v4, Zustand, OKLCH color space, lucide-react icons

**Spec:** `docs/superpowers/specs/2026-04-09-dark-mode-design.md`

```typescript
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

export type ThemePreference = 'light' | 'dark' | 'system'

interface ThemeState {
  preference: ThemePreference
  setPreference: (preference: ThemePreference) => void
}

export const useThemeStore = create<ThemeState>()(
  persist(
    (set) => ({
      preference: 'system',
      setPreference: (preference) => set({ preference }),
    }),
    { name: 'theme-preference' },
  ),
)
```

Create `src/lib/theme.ts`:

```typescript
import type { ThemePreference } from '@/stores/theme-store'

export function getEffectiveTheme(
  preference: ThemePreference,
  systemPrefersDark: boolean,
): 'light' | 'dark' {
  if (preference === 'system') return systemPrefersDark ? 'dark' : 'light'
  return preference
}

export function applyTheme(effectiveTheme: 'light' | 'dark'): void {
  if (effectiveTheme === 'dark') {
    document.documentElement.classList.add('dark')
  } else {
    document.documentElement.classList.remove('dark')
  }
}
```

### Task 4: Flash Prevention & App Integration

**Files:**

- [ ] **Step 1: Add flash-prevention script to index.html**

Add an inline script in `<head>` that reads localStorage and applies `.dark` before the page renders:

```html
<!doctype html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <link rel="icon" href="/favicon.ico" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Adare Web</title>
    <script>
      (function() {
        var stored = localStorage.getItem('theme-preference');
        var pref = 'system';
        if (stored) {
          try { pref = JSON.parse(stored).state.preference; } catch(e) {}
        }
        var isDark = pref === 'dark' ||
          (pref === 'system' && window.matchMedia('(prefers-color-scheme: dark)').matches);
        if (isDark) document.documentElement.classList.add('dark');
      })();
    </script>
  </head>
  <body>
    <div id="root"></div>
    <script type="module" src="/src/main.tsx"></script>
  </body>
</html>
```

- [ ] **Step 2: Hook up theme applicator in App.tsx**

Add a `useEffect` in `App.tsx` that subscribes to the theme store and applies theme changes, including listening for OS preference changes:

```typescript
import { useEffect } from 'react'
import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
import { RouterProvider, createRouter } from '@tanstack/react-router'
import { routeTree } from '@/routes'
import { useThemeStore } from '@/stores/theme-store'
import { applyTheme, getEffectiveTheme } from '@/lib/theme'

const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 30_000,
      retry: 1,
    },
  },
})

const router = createRouter({
  routeTree,
  context: { queryClient },
})

declare module '@tanstack/react-router' {
  interface Register {
    router: typeof router
  }
}

export function App() {
  const preference = useThemeStore((s) => s.preference)

  useEffect(() => {
    const mq = window.matchMedia('(prefers-color-scheme: dark)')
    const apply = () => applyTheme(getEffectiveTheme(preference, mq.matches))

    apply()
    mq.addEventListener('change', apply)
    return () => mq.removeEventListener('change', apply)
  }, [preference])

  return (
    <QueryClientProvider client={queryClient}>
      <RouterProvider router={router} />
    </QueryClientProvider>
  )
}
```

```typescript
function ThemeSubmenu() {
  const { preference, setPreference } = useThemeStore()

  const options: { value: ThemePreference; label: string; icon: React.ReactNode }[] = [
    { value: 'light', label: 'Light', icon: <Sun className="w-4 h-4" /> },
    { value: 'dark', label: 'Dark', icon: <Moon className="w-4 h-4" /> },
    { value: 'system', label: 'System', icon: <Monitor className="w-4 h-4" /> },
  ]

  return (
    <div className="px-1 py-1 border-t">
      <div className="px-3 py-1.5 text-xs font-medium text-muted-foreground">Theme</div>
      {options.map((opt) => (
        <button
          key={opt.value}
          onClick={() => setPreference(opt.value)}
          className="flex items-center gap-2 w-full text-left px-3 py-1.5 text-sm text-popover-foreground hover:bg-accent rounded"
        >
          {opt.icon}
          {opt.label}
          {preference === opt.value && <Check className="w-3 h-3 ml-auto text-primary" />}
        </button>
      ))}
    </div>
  )
}
```
